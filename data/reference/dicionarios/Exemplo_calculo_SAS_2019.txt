LIBNAME PENSE "";


/* SEPARA TOTAIS DOS DOMÍNIOS DE PÓS-ESTRATIFICAÇÃO */
PROC SQL;
CREATE TABLE TOTAIS_POSEST AS
SELECT DISTINCT POSEST, TOTAIS_POSEST AS _PSTOTAL_
FROM PENSE.PENSE2019_MICRODADOS;
QUIT;


/* EXEMPLO DE ESTIMAÇÃO COM PÓS-ESTRATIFICAÇÃO */
PROC SURVEYMEANS DATA=PENSE.PENSE2019_MICRODADOS  SUM CLSUM CVSUM ;

WEIGHT PESO_INICIAL;
/* Estrato de estimação */
STRATA ESTRATO;
/* definição da variável do conglomerado primário */
CLUSTER ESCOLA;
/* INCLUINDO PÓS-ESTRATO */
POSTSTRATA POSEST / PSTOTAL=TOTAIS_POSEST;
/* Definição das variáveis tratadas como categóricas */
CLASS UF;
/* Variáveis a serem tabuladas */
VAR UF;
/* Domínio de estimação */
DOMAIN IND_EXPANSAO;
/* Gravação dos arquivos de saída */
ODS OUTPUT DOMAIN = DOMINIOS (WHERE = (IND_EXPANSAO = "1"));

RUN;
